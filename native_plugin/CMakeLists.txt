cmake_minimum_required(VERSION 3.28)
project(sternenflug VERSION 0.1.2 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(ObsRuntimeSDK)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets)

file(GLOB PLUGIN_SOURCES CONFIGURE_DEPENDS "src/*.cpp" "src/*.h" "src/*.hpp")
list(FILTER PLUGIN_SOURCES EXCLUDE REGEX "_tests?\\.cpp$")
if(NOT PLUGIN_SOURCES)
  message(FATAL_ERROR "No plugin sources found in src/.")
endif()

# Optional real-GPU preview exporter. Never part of a normal plugin build.
option(STERNENFLUG_BUILD_PREVIEW "Build the isolated GPU preview exporter" OFF)
if(STERNENFLUG_BUILD_PREVIEW)
  add_executable(sf_preview_capture EXCLUDE_FROM_ALL
    tests/preview_capture.cpp src/renderer.cpp src/core.cpp)
  target_include_directories(sf_preview_capture PRIVATE src)
  target_link_libraries(sf_preview_capture PRIVATE OBS::libobs)
  target_compile_definitions(sf_preview_capture PRIVATE
    NOMINMAX WIN32_LEAN_AND_MEAN STERNENFLUG_VERSION="${PROJECT_VERSION}")
  set_target_properties(sf_preview_capture PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
  if(MSVC)
    target_compile_options(sf_preview_capture PRIVATE /W4 /utf-8 /EHsc)
  endif()
endif()
add_library(sternenflug MODULE ${PLUGIN_SOURCES})
target_link_libraries(sternenflug PRIVATE OBS::libobs OBS::frontend Qt6::Core Qt6::Gui Qt6::Widgets)
target_compile_definitions(sternenflug PRIVATE
  NOMINMAX WIN32_LEAN_AND_MEAN
  STERNENFLUG_VERSION="${PROJECT_VERSION}"
  )
target_include_directories(sternenflug PRIVATE src)
if(MSVC)
  target_compile_options(sternenflug PRIVATE /W4 /permissive- /utf-8 /EHsc)
endif()
set_target_properties(sternenflug PROPERTIES
  PREFIX ""
  OUTPUT_NAME sternenflug
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")

install(TARGETS sternenflug RUNTIME DESTINATION sternenflug/bin/64bit LIBRARY DESTINATION sternenflug/bin/64bit)
install(DIRECTORY data/ DESTINATION sternenflug/data OPTIONAL)

include(CTest)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/core_tests.cpp")
  add_executable(sf_core_tests src/core_tests.cpp src/core.cpp)
  target_include_directories(sf_core_tests PRIVATE src)
  set_target_properties(sf_core_tests PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
  if(MSVC)
    target_compile_options(sf_core_tests PRIVATE /W4 /permissive- /utf-8 /EHsc /UNDEBUG)
  endif()
  add_test(NAME sf_core_tests COMMAND sf_core_tests)
  set_tests_properties(sf_core_tests PROPERTIES TIMEOUT 60)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/renderer_smoke.cpp")
  add_executable(sf_renderer_smoke tests/renderer_smoke.cpp src/renderer.cpp src/core.cpp)
  target_include_directories(sf_renderer_smoke PRIVATE src)
  target_link_libraries(sf_renderer_smoke PRIVATE OBS::libobs)
  target_compile_definitions(sf_renderer_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(sf_renderer_smoke PRIVATE /W4 /utf-8 /EHsc /UNDEBUG)
  endif()
  add_test(NAME sf_renderer_smoke COMMAND sf_renderer_smoke)
  set_tests_properties(sf_renderer_smoke PROPERTIES TIMEOUT 90)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/editor_smoke.cpp")
  add_executable(sf_editor_smoke tests/editor_smoke.cpp src/editor.cpp src/codec.cpp src/core.cpp src/editor.hpp)
  target_include_directories(sf_editor_smoke PRIVATE src)
  target_link_libraries(sf_editor_smoke PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
  target_compile_definitions(sf_editor_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(sf_editor_smoke PRIVATE /W4 /utf-8 /EHsc /UNDEBUG)
  endif()
  add_test(NAME sf_editor_smoke COMMAND sf_editor_smoke)
  set_tests_properties(sf_editor_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal" TIMEOUT 90)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/plugin_smoke.cpp")
  add_executable(sf_plugin_smoke tests/plugin_smoke.cpp)
  target_include_directories(sf_plugin_smoke PRIVATE src)
  target_link_libraries(sf_plugin_smoke PRIVATE OBS::libobs OBS::frontend)
  target_compile_definitions(sf_plugin_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(sf_plugin_smoke PRIVATE /W4 /utf-8 /EHsc /UNDEBUG)
  endif()
  add_dependencies(sf_plugin_smoke sternenflug)
  add_test(NAME sf_plugin_smoke COMMAND sf_plugin_smoke "$<TARGET_FILE:sternenflug>")
  set_tests_properties(sf_plugin_smoke PROPERTIES TIMEOUT 90)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
  add_subdirectory(tests)
endif()

# Reproducible offscreen latency benchmark. Not a CTest or release-DLL target.
option(STERNENFLUG_BUILD_BENCHMARK "Build the isolated renderer benchmark" OFF)
if(STERNENFLUG_BUILD_BENCHMARK)
  add_executable(sf_renderer_benchmark EXCLUDE_FROM_ALL
    tests/renderer_benchmark.cpp src/renderer.cpp src/core.cpp)
  target_include_directories(sf_renderer_benchmark PRIVATE src)
  target_link_libraries(sf_renderer_benchmark PRIVATE OBS::libobs)
  if(WIN32)
    target_link_libraries(sf_renderer_benchmark PRIVATE dxgi)
  endif()
  target_compile_definitions(sf_renderer_benchmark PRIVATE
    NOMINMAX WIN32_LEAN_AND_MEAN STERNENFLUG_VERSION="${PROJECT_VERSION}")
  set_target_properties(sf_renderer_benchmark PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
  if(MSVC)
    target_compile_options(sf_renderer_benchmark PRIVATE /W4 /utf-8 /EHsc)
  endif()
endif()
